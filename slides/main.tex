\section{Цель и задачи}
% ===========================================================================

% --- Slide: Цель и задачи ВКР --------------------------------------------
\begin{frame}{Цель: построить каскад дешевле прямой LLM и доказать его качество}
\fontsize{11pt}{13pt}\selectfont
\textbf{Цель.} Программное средство обогащения товарных данных на основе
гибридной каскадной архитектуры.

\vspace{0.5em}
\textbf{Частные задачи}
\begin{enumerate}
    \setlength{\itemsep}{0.2em}
    \item Разработать архитектуру гибридного каскадного конвейера (5 слоёв) с поатрибутной политикой маршрутизации.
    \item Реализовать каждый слой как самостоятельный программный компонент с независимым обновлением.
    \item Построить эталонную разметку и тестовую выборку без пересечения товарных кодов (code-disjoint).
    \item Провести экспериментальную оценку: точность, покрытие, стоимость; сравнение с прямым вызовом БЯМ.
    \item Разработать демонстрационный программный комплекс с REST API и веб-интерфейсом.
\end{enumerate}
\end{frame}

% --- Slide: Объект, предмет, рабочее предположение -----------------------
\begin{frame}{Главная гипотеза: каскад дешевле и не теряет в точности}
\fontsize{11pt}{14pt}\selectfont
\textbf{Объект.} Процессы обогащения карточек товаров в интернет-магазинах.

\vspace{0.5em}
\textbf{Предмет.} Методы и алгоритмы извлечения атрибутов, политики
маршрутизации между слоями разной стоимости и протоколы их оценки.

\vspace{0.8em}
\textbf{Рабочее предположение.}\\[0.2em]
\fontsize{10pt}{13pt}\selectfont
\begin{itemize}
    \setlength{\itemsep}{0.15em}
    \item \textbf{Каскад} из слоёв разной стоимости (regex $\to$ ML $\to$ Bayes $\to$ LLM)
          обеспечивает заданную точность обогащения товарных данных.
    \item \textbf{Поатрибутная политика эскалации} по калиброванной уверенности
          кратно сокращает стоимость относительно прямого вызова большой языковой модели.
    \item \textbf{Локальное развёртывание} возможно на коммерчески доступной инфраструктуре.
\end{itemize}
\end{frame}

\section{Постановка}
% ===========================================================================

% --- Slide: Формальная постановка задачи (вариант A — полная формализация) -
\begin{frame}{Задача формально: подобрать пороги $\tau_{i,a}$ при бюджете LLM $\le 10\,\%$}
\fontsize{10pt}{12pt}\selectfont
\begin{columns}[T]
\column{0.49\textwidth}
\textbf{Дано.} Товар $x = (\texttt{name}, \texttt{brand}, \texttt{ingr}, \texttt{qty}, \texttt{code})$;
множество атрибутов $A$, $|A|=21$ (паста 8 / шоколад 7 / сыры 6); для каждого $a$ — словарь
$\mathcal{Y}_a$ с отказом $\bot$.

\vspace{0.5em}
\textbf{Политика обогащения.}
\[
  \pi_a \;=\; f_4 \,\circ\, \sigma_3 \,\circ\, f_3 \,\circ\, \sigma_2 \,\circ\, f_2 \,\circ\, \sigma_1 \,\circ\, f_1
\]
\fontsize{9pt}{11pt}\selectfont
$f_i(x) \to (\hat y, c)$ — слой $i$ возвращает значение и уверенность;
$\sigma_i$ — ворота: принять $\hat y$ при $c \ge \tau_{i,a}$, иначе передать дальше.

\column{0.49\textwidth}
\fontsize{10pt}{12pt}\selectfont
\textbf{Оптимизационная задача.}
\[
  \max_{\{\tau_{i,a}\}} \;\; \tfrac{1}{|A|}\textstyle\sum_{a} \mathrm{acc}^{e2e}_a
  \;\;\text{при}\;\;
  \mathrm{cost}_{\mathrm{LLM}} \le \beta = 10\,\%.
\]

\vspace{0.5em}
\fontsize{9.5pt}{11.5pt}\selectfont
\textbf{Ограничения}
\begin{itemize}
    \setlength{\itemsep}{0.1em}
    \item признаки только из партнёрских полей (без \texttt{categories\_tags} — анти-утечка);
    \item проверка на непересекающихся товарных кодах;
    \item локальное развёртывание возможно (слой 4 — open-model).
\end{itemize}
\end{columns}
\end{frame}

% --- Slide: Состав требуемых функций -------------------------------------
\begin{frame}{Требования: 6 функций + 4 нефункциональных + 3 архитектурных}
\fontsize{10pt}{12pt}\selectfont
\begin{columns}[T]
\column{0.49\textwidth}
\textbf{Функциональные (что делает)}
\begin{itemize}
    \setlength{\itemsep}{0.15em}
    \item \textbf{F1.} Значение каждого атрибута
    \item \textbf{F2.} Слой-источник предсказания
    \item \textbf{F3.} Калиброванная уверенность $c \in [0,1]$
    \item \textbf{F4.} Эскалация на запасной слой
    \item \textbf{F5.} Байес-валидация подозрительных решений
    \item \textbf{F6.} Поатрибутная настройка через конфиг
\end{itemize}

\column{0.49\textwidth}
\textbf{Нефункциональные (как работает)}
\begin{itemize}
    \setlength{\itemsep}{0.15em}
    \item \textbf{N1.} Доля LLM $\le 10\,\%$ ячеек
    \item \textbf{N2.} Локальное развёртывание (без внешних API)
    \item \textbf{N3.} Обобщение на новых товарных кодах
    \item \textbf{N4.} Воспроизводимость через \texttt{reproduce.sh}
\end{itemize}

\vspace{0.3em}
\textbf{Архитектурные (как устроено)}\\
\fontsize{9.5pt}{11pt}\selectfont
\textbf{A1.}~\texttt{docker-compose}; \textbf{A2.}~REST-контракт между слоями; \textbf{A3.}~холодный старт без правки ядра.
\end{columns}
\end{frame}

% --- Slide: Система качества и интегральный критерий ---------------------
\begin{frame}{Критерий качества $Q$ — 4 показателя в свёртке с весами}
\fontsize{9pt}{11pt}\selectfont
Качество описано \textbf{четырьмя показателями} (точность, стоимость, покрытие,
калибровка), сведёнными в интегральный критерий $Q$:
\[
    Q \;=\; w_1\cdot\mathrm{Acc}
        \;+\; w_2\cdot\Bigl(1 - \tfrac{C_{\mathrm{LLM}}}{C_{\max}}\Bigr)
        \;+\; w_3\cdot\mathrm{Cov}
        \;-\; w_4\cdot\mathrm{ECE}
\]

\vspace{-0.2em}
\fontsize{8.5pt}{10.5pt}\selectfont
$\mathrm{Acc}$ — точность на непересекающихся товарных кодах (code-disjoint); $C_{\mathrm{LLM}}/C_{\max}$ — доля LLM-вызовов;
$\mathrm{Cov}$ — покрытие каскадом до LLM; $\mathrm{ECE}$ — ошибка калибровки.\\
\textbf{Веса:} $w_1 = 0{,}40$, $w_2 = 0{,}30$, $w_3 = 0{,}20$, $w_4 = 0{,}10$
(приоритет точности над стоимостью согласован с типовыми требованиями PIM-индустрии).

\vspace{0.3em}
\centering
\fontsize{8.5pt}{10.5pt}\selectfont
\setlength{\tabcolsep}{6pt}
\renewcommand{\arraystretch}{1.05}
\begin{tabular}{lcc}
\toprule
\textbf{Показатель} & \textbf{Целевое} & \textbf{Достигнуто} \\
\midrule
Точность каскада                  & $\ge 90\,\%$  & \pos{95{,}5\,\%} \\
$C_{\mathrm{LLM}}/C_{\max}$       & $\le 0{,}10$  & \pos{0{,}041} \\
Cov (покрытие до LLM)             & $\ge 0{,}90$  & \pos{0{,}959} \\
ECE (калибровка)                  & $\le 0{,}05$  & per-attr ниже целевого \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{B0. Метрики и обозначения (для вопросов)}
\fontsize{10.5pt}{13pt}\selectfont
\textbf{AND-условие слоя 2.} Предсказание $\hat y$ принимается
$\Longleftrightarrow P_{\mathrm{ML}}(\hat y \mid x) \ge \tau_a \;\land\; P_B(\hat y \mid \mathbf{e}) \ge \theta_a$
— формализация ансамблевой оценки уверенности как конъюнкции порогов
ML-классификатора ($\tau_a$) и байесовского валидатора ($\theta_a$).

\vspace{0.6em}
\textbf{Метрики:}
\begin{itemize}
    \fontsize{9.5pt}{12pt}\selectfont
    \setlength{\itemsep}{0.15em}
    \item $\mathrm{acc}^{\mathrm{cov}}_k = \frac{|\hat y_k = y_k \land \hat y_k \ne \mathrm{null}|}{|\hat y_k \ne \mathrm{null}|}$ — точность на покрытой части (только по ячейкам, где $\hat y \ne \bot$);
    \item $\mathrm{cov}_k = \frac{|\hat y_k \ne \mathrm{null}|}{N}$ — покрытие, доля ячеек с ненулевым предсказанием;
    \item $\mathrm{acc}^{\mathrm{e2e}}_k = \mathrm{acc}^{\mathrm{cov}}_k \cdot \mathrm{cov}_k$ — сквозная точность (отказ $\bot$ засчитывается как ошибка), production-realistic;
    \item $\mathrm{cost}_{\mathrm{LLM}}$ — ожидаемая доля вызовов запасного слоя на 1000 ячеек (бюджет $\beta = 10\,\%$).
\end{itemize}

\vspace{0.3em}
\fontsize{9.5pt}{12pt}\selectfont\itshape\color{black!60}
Связь $\mathrm{acc}^{\mathrm{e2e}} = \mathrm{acc}^{\mathrm{cov}} \cdot \mathrm{cov}$
объясняет разрыв между только-каскадной точностью (95,5\,\%) и сквозной
с маршрутизатором (93,0\,\%): часть ячеек теряется на $\mathrm{cov} < 1$.
\end{frame}

\textbf{Цикл «аудит — правки экстрактора — переобучение — измерение»} воспроизведён
для всех 3 категорий с приростом \pos{+18{,}2}, \pos{+16{,}3}, \pos{+14{,}4}\,пп.

\vspace{0.3em}
\textbf{Калибровка уверенности} — комбинация Platt + изотонической регрессии
поатрибутно; ошибка ECE измеряется на отдельной выборке.
\end{frame}

\begin{frame}{B4. Утечки и борьба с ними}
\fontsize{10pt}{12pt}\selectfont
\textbf{Запрет утечки \texttt{categories\_tags} в эмбеддинги.}

Эмбеддинги Слоя~2 строятся \textbf{только} из партнёрских полей:
\texttt{product\_name + brands + ingredients\_text + quantity}.
\texttt{categories\_tags} — это \textit{выход} обогащения, не вход;
использование как признака привело бы к утечке целевой переменной.

\vspace{0.3em}
\textbf{Разбиение по непересекающимся товарным кодам (60/20/20; brand overlap train↔eval 82–85\%).}
Бренды в обучающей, валидационной и контрольной выборках не пересекаются —
гарантия обобщения на новые бренды.

\vspace{0.3em}
\textbf{Анализ по расстоянию k-NN (§3.3.7.3):} точность \textbf{не падает} на удалённых
от обучающей выборки товарах — монотонный рост от Q1 (82{,}7\,\%) до Q4 (91{,}6\,\%).
\end{frame}

\begin{frame}{B6. Стоимость языковой модели на 1\,млн SKU}
\fontsize{10pt}{12pt}\selectfont
\textbf{Каскад делегирует языковой модели лишь $4{,}1$\,\% ячеек} ($n=17\,062$) —
прямая прокси к стоимости. Сокращение затрат \textbf{95{,}9\,\% (в $24\times$ раз)}
относительно «прямой языковой модели на всё».

\vspace{0.3em}
\textbf{Распределение по слоям:}
\fontsize{9pt}{11pt}\selectfont
L1 (regex высокой точности) — 21{,}9\,\%, L2 (ML) — \textbf{73{,}2\,\%},
L3 (regex низкой точности) — 0{,}8\,\%, L4 (языковая модель) — 4{,}1\,\%.

\vspace{0.3em}
\fontsize{10pt}{12pt}\selectfont
\textbf{Проекция на 1\,млн SKU × 20 атрибутов} ($\sim 2{,}0\cdot 10^7$ ячеек):
\begin{itemize}
    \fontsize{9pt}{11pt}\selectfont
    \setlength{\itemsep}{0.15em}
    \item Прямая модель — оплачены все $\sim 2{,}0\cdot 10^7$ вызовов.
    \item Каскад — оплачены лишь $\sim 8{,}2\cdot 10^5$ вызовов ($4{,}1\,\%$).
\end{itemize}

\vspace{0.2em}
\textbf{Топ-3 затратных атрибутов для языковой модели:}
pasta.grain\_type (32{,}9\,\%), pasta.pasta\_shape (22{,}8\,\%), cheeses.is\_pdo (13{,}6\,\%).
\end{frame}
